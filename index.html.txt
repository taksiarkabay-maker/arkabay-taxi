    <script>
        // Инициализация Telegram WebApp
        const tg = window.Telegram.WebApp;
        tg.expand(); // Развернуть на весь экран

        // Инициализация карты: координаты центра между Аркабай и Панфилово, масштаб 13
        const map = L.map('map').setView([43.3762, 77.0850], 13);
        L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);

        // Добавим маркеры для ориентира (опционально, чтобы пользователь сразу видел ключевые точки)
        L.marker([43.3931, 77.0701]).addTo(map).bindPopup('п. Аркабай');
        L.marker([43.3593, 77.1000]).addTo(map).bindPopup('п. Панфилово');

        // Логика кнопки заказа
        document.getElementById('orderBtn').addEventListener('click', () => {
            const fromText = document.getElementById('from').value;
            const toText = document.getElementById('to').value;
            const tariff = document.getElementById('tariff').value;

            if (!fromText || !toText) {
                alert('Пожалуйста, заполните адреса!');
                return;
            }

            // Формируем данные заказа
            const orderData = {
                from: fromText,
                to: toText,
                tariff: tariff,
                price: Math.floor(Math.random() * 300) + 400 // Симуляция расчета цены для этой дистанции
            };

            // Отправляем данные обратно в бота Telegram
            tg.sendData(JSON.stringify(orderData));
            tg.close();
        });
    </script>
